#   python3 -m venv .venv && source .venv/bin/activate   # fish: activate.fish
#   pip install -r requirements.txt                      # 锁版本装依赖
#   pip install -e .                                     # 安装本包；不装则运行时带 PYTHONPATH=.

#   且 MuJoCo 查看器强依赖的两个（glfw / PyOpenGL）。
